\begin{itemize}
	\item LTSs are finite state machines used to graphically represent the FSP description of a process.
	\begin{itemize}
		\item Starting states are denoted as a circle with a dot in the middle.
	\end{itemize}
\end{itemize}

\begin{example}
	Represent the following FSP as an LTS.
	\begin{align*}
		\mathtt{A} = (\mathtt{a} \rightarrow \mathtt{B} \; | \; \mathtt{b} \rightarrow \mathtt{C}) \\
		\mathtt{B} = (\mathtt{a} \rightarrow \mathtt{B} \; | \; \mathtt{b} \rightarrow \mathtt{D}) \\
		\mathtt{C} = (\mathtt{d} \rightarrow \mathtt{C}) \\
		\mathtt{D} = (\mathtt{a} \rightarrow \mathtt{C} \; | \; \mathtt{c} \rightarrow \mathtt{B}) \\
	\end{align*}
\end{example}

\begin{itemize}
	\item FSPs can nest processes within other processes (such as through processes substitution) to make FSPs more concise.
	\begin{itemize}
		\item Directly translating an FSP with nested processes into an LTS will yield an incorrect representation of the FSP.
	\end{itemize}
	\item An FSP is said to be put into \emph{elementary/normal form} if all nested processes in the FSP are expanded out into separate processes.
	\begin{itemize}
		\item FSPs must be put into elementary form to be correctly represented as an LTS.
	\end{itemize}
\end{itemize}

\begin{example}
	Represent the following FSP as an LTS.
	\begin{align*}
		\mathtt{A} &= (\mathtt{a} \rightarrow \mathtt{b} \rightarrow \mathtt{B} \; | \; \mathtt{b} \rightarrow (\mathtt{a} \rightarrow \mathtt{c} \rightarrow \mathtt{A} \; | \; \mathtt{b} \rightarrow \mathtt{B})) \\
		\mathtt{B} &= (\mathtt{a} \rightarrow \mathtt{c} \rightarrow (\mathtt{a} \rightarrow \mathtt{A} \; | \; \mathtt{b} \rightarrow \mathtt{B}))
	\end{align*}
\end{example}

\begin{solution}
	First, we must put our FSP into normal form.
	
	\begin{align*}
		\mathtt{A} &= (\mathtt{a} \rightarrow \mathtt{B_1} \; | \; \mathtt{b} \rightarrow \mathtt{D}) \\
		\mathtt{B} &= (\mathtt{a} \rightarrow \mathtt{A_2}) \\
		\mathtt{D} &= (\mathtt{a} \rightarrow \mathtt{A_1} \; | \; \mathtt{b} \rightarrow \mathtt{B}) \\
		\mathtt{A_1} &= (\mathtt{c} \rightarrow \mathtt{A}) \\
		\mathtt{A_2} &= (\mathtt{c} \rightarrow \mathtt{B_2}) \\
		\mathtt{B_1} &= (\mathtt{b} \rightarrow \mathtt{B}) \\
		\mathtt{B_2} &= (\mathtt{a} \rightarrow \mathtt{A} \; | \; \mathtt{b} \rightarrow \mathtt{B}) \\
	\end{align*}
	
	Now that our FSP is in elementary form, we can represent it as:

		\caption{LTS representation of the given FSP.}
	\end{figure}
\end{solution}
